cmake_minimum_required(VERSION 3.20)
project(june_whisper_worker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT JUNE_WHISPER_SOURCE)
  message(FATAL_ERROR "JUNE_WHISPER_SOURCE must name the pinned, already built whisper.cpp installation")
endif()
# Link the existing installation, never add_subdirectory/rebuild Whisper or
# download a library. speech:setup owns its pinned CPU-only static build.
set(ENGINE_BUILD "${JUNE_WHISPER_SOURCE}/build")
file(READ "${ENGINE_BUILD}/CMakeCache.txt" ENGINE_CACHE)
foreach(option GGML_METAL GGML_CUDA WHISPER_COREML BUILD_SHARED_LIBS)
  if(ENGINE_CACHE MATCHES "${option}:BOOL=ON")
    message(FATAL_ERROR "The installed runtime must be CPU-only and static: ${option}=ON")
  endif()
endforeach()
find_package(Threads REQUIRED)
find_library(WHISPER_LIBRARY whisper PATHS "${ENGINE_BUILD}/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_LIBRARY ggml PATHS "${ENGINE_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_CPU_LIBRARY ggml-cpu PATHS "${ENGINE_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_BASE_LIBRARY ggml-base PATHS "${ENGINE_BUILD}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_BLAS_LIBRARY ggml-blas PATHS "${ENGINE_BUILD}/ggml/src/ggml-blas" NO_DEFAULT_PATH)
add_executable(june-whisper-worker ../whisper-worker.cpp)
target_include_directories(june-whisper-worker PRIVATE
  "${JUNE_WHISPER_SOURCE}/include" "${JUNE_WHISPER_SOURCE}/ggml/include" "${JUNE_WHISPER_SOURCE}/examples")
target_compile_options(june-whisper-worker PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(june-whisper-worker PRIVATE "${WHISPER_LIBRARY}" "${GGML_LIBRARY}" "${GGML_CPU_LIBRARY}")
if(GGML_BLAS_LIBRARY)
  target_link_libraries(june-whisper-worker PRIVATE "${GGML_BLAS_LIBRARY}")
  if(NOT APPLE)
    find_package(BLAS REQUIRED)
    target_link_libraries(june-whisper-worker PRIVATE BLAS::BLAS)
  endif()
endif()
target_link_libraries(june-whisper-worker PRIVATE "${GGML_BASE_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
if(APPLE)
  find_library(ACCELERATE_FRAMEWORK Accelerate REQUIRED)
  target_link_libraries(june-whisper-worker PRIVATE "${ACCELERATE_FRAMEWORK}")
else()
  target_link_libraries(june-whisper-worker PRIVATE m)
endif()
